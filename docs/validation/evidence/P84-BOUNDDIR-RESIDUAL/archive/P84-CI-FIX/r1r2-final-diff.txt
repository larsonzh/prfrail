+func splitDosPathComponents(path string) []string {

+// directoryEntryShortName reads the 8.3 short name of one directory entry OUT OF
+// THE VERIFIED PARENT HANDLE's own enumeration, and binds the entry to the object
+// the caller already opened: the entry whose long name equals longName must carry
+// the same file id as the opened object (wantHigh/wantLow), and the short name is
+// read from that SAME entry. No spelling derived from an absolute path participates
+// in any admission decision, and an entry that is missing, that no longer belongs
+// to the opened object, that is malformed, or that declares no short name yields
+// ("", false) - every shape refuses fail-closed.
+func directoryEntryShortName(parent syscall.Handle, longName string, wantHigh, wantLow uint32) (string, bool) {
+	if err := assertFileIDBothDirInfoLayout(); err != nil {
+		return "", false
+	}
+	restart := true
+	for size := uint32(readDirBufferStart); ; {
+		buffer := make([]byte, size)
+		// The first call of the scan restarts it; the continuation calls use the
+		// non-restart class so they resume where the previous buffer ended - the
+		// restart class on every call would restart the scan forever.
+		class := uintptr(fileIdBothDirectoryInfo)
+		if restart {
+			class = uintptr(fileIdBothDirectoryInfoRestart)
+		}
+		result, _, callErr := procGetFileInformationByHandleEx.Call(
+			uintptr(parent),
+			class,
+			uintptr(unsafe.Pointer(&buffer[0])),
+			uintptr(size),
+		)
+		if result == 0 {
+			errno, _ := callErr.(syscall.Errno)
+			switch errno {
+			case syscall.ERROR_NO_MORE_FILES, syscall.ERROR_FILE_NOT_FOUND:
+				// The directory is exhausted without the name: no evidence.
+				return "", false
+			case syscall.ERROR_MORE_DATA, syscall.ERROR_INSUFFICIENT_BUFFER, syscall.ERROR_BUFFER_OVERFLOW:
+				if size >= readDirBufferLimit {
+					return "", false
+				}
+				size *= 2
+				continue
+			}
+			return "", false
+		}

+		restart = false

+		if refuse || evidenced {

+				return "", false
+			}

+		// scan where the handle position stopped.

+// scanFileIDBothDirInfoForShortName is the pure, testable half of the directory
+// entry evidence. It walks one FILE_ID_BOTH_DIR_INFORMATION buffer and reports:
+//
+//   - evidenced: the entry whose long name equals longName was found, its full
+//     64-bit file id equals the opened object's, and it declares a short name
+//     (returned byte for byte);
+//   - refuse: the buffer is malformed - a NextEntryOffset that is not 8-byte
+//     aligned, shorter than one header, or past the buffer; a declared record that
+//     does not fit the buffer; a FileNameLength that is zero or odd; or a NUL unit
+//     inside either declared name range. A malformed answer must never be parsed
+//     on a guess, so refuse fails the whole admission closed;
+//   - neither: the buffer carries no matching entry (the scan continues in the
+//     caller), or the matching entry belongs to another object or declares no
+//     short name (no evidence, never an error the caller can use).
+//
+// The terminal NUL unit of a declared ShortName range is allowed and stripped:
+// the fixed ShortName array is NUL-terminated by convention when the name does not
+// fill it, and some filesystems count the terminator in ShortNameLength (the same
+// shape N14 pins for reparse names).
+func scanFileIDBothDirInfoForShortName(buffer []byte, longName string, wantHigh, wantLow uint32) (shortName string, evidenced, refuse bool) {
+	nameOffset := unsafe.Offsetof(fileIDBothDirInfo{}.FileName)
+	bufferBase := unsafe.Pointer(&buffer[0])
+	for offset := uintptr(0); ; {
+		if offset+nameOffset > uintptr(len(buffer)) {
+			return "", false, true
+		}
+		entry := (*fileIDBothDirInfo)(unsafe.Add(bufferBase, offset))
+		next := uintptr(entry.NextEntryOffset)
+		if next != 0 {
+			// NextEntryOffset is validated before the record is read further: it
+			// must be 8-byte aligned (the kernel's LONGLONG alignment), at least
+			// one full header long, and leave a full header at its target.
+			if next%8 != 0 || next < nameOffset || offset+next+nameOffset > uintptr(len(buffer)) {
+				return "", false, true
+			}
+		}
+		// This record's declared range must fit the buffer and be a legal UTF-16
+		// byte length: zero and odd lengths cannot name a directory entry.
+		if entry.FileNameLength == 0 || entry.FileNameLength%2 != 0 ||
+			offset+nameOffset+uintptr(entry.FileNameLength) > uintptr(len(buffer)) {
+			return "", false, true
+		}
+		units := uintptr(entry.FileNameLength) / 2
+		nameUnits := unsafe.Slice((*uint16)(unsafe.Add(bufferBase, offset+nameOffset)), units)
+		for _, unit := range nameUnits {
+			if unit == 0 {
+				// A NUL inside the declared name range would decode a prefix of
+				// the stored name and the entry could be matched on that prefix.
+				return "", false, true
+			}
+		}
+		if syscall.UTF16ToString(nameUnits) != longName {
+			if next == 0 {
+				return "", false, false
+			}
+			offset += next
+			continue
+		}
+		// The entry is only evidence when it still belongs to the object the
+		// caller opened: the full 64-bit file id must match.
+		if uint64(entry.FileId) != uint64(wantHigh)<<32|uint64(wantLow) {
+			return "", false, false
+		}
+		if entry.ShortNameLength == 0 {
+			// Found and bound, but no short name: the divergence is not an 8.3
+			// alias, so there is no evidence.
+			return "", false, false
+		}
+		if entry.ShortNameLength%2 != 0 || uintptr(entry.ShortNameLength)/2 > uintptr(len(entry.ShortName)) {
+			return "", false, true
+		}
+		shortUnits := uintptr(entry.ShortNameLength) / 2
+		shortNameUnits := entry.ShortName[:shortUnits]
+		// The terminal NUL unit of the fixed array is the terminator, not part of
+		// the name; an embedded NUL refuses (the same discipline as the reparse
+		// name scan, N14).
+		if shortUnits > 0 && shortNameUnits[shortUnits-1] == 0 {
+			shortNameUnits = shortNameUnits[:shortUnits-1]
+		}
+		for _, unit := range shortNameUnits {
+			if unit == 0 {
+				return "", false, true
+			}
+		}
+		return syscall.UTF16ToString(shortNameUnits), true, false
+	}

+// when the stored target's spelling does not byte-match the bound root's
+// handle-derived name: a stored target whose prefix spells the bound root's own
+// path through 8.3 short-name aliases (the CI windows runner shape, where the
+// user profile component is stored as its short alias while the handle names the
+// long spelling). Only what the OS proves to be a root-prefix 8.3 alias is
+// admitted, and never for reading directly:

+//     below it are returned as the suffix byte for byte, unwritten;
+//   - the candidate root is opened one component at a time, every step relative to
+//     the previously verified parent handle (the volume root, opened once, is the

+//     handle's own volume serial and file index, never an ancestor's.

+// suffix walk that follows runs from the bound handle and never uses the candidate
+// path again, so an alias changed afterwards cannot redirect it (fail-closed).

+// p84ShortPathName reports the 8.3 short spelling of one existing path.
+func p84ShortPathName(path string) (string, error) {
+	ptr, err := syscall.UTF16PtrFromString(path)
+	if err != nil {

+	}

+// p84DirEntryBuffer builds one FILE_ID_BOTH_DIR_INFORMATION record in an
+// 8-byte-padded byte buffer for the focused scanner assertions. The FileName
+// range is written WITHOUT a terminator (FileNameLength is authoritative), and
+// the ShortName range is written exactly as many units as declared - variants
+// mutate the buffer afterwards.
+func p84DirEntryBuffer(name, shortName string, fileID uint64, next uint32) []byte {
+	// utf16.Encode returns exactly the code units of the name, no terminator:
+	// FileNameLength and ShortNameLength are authoritative, so nothing is
+	// stripped here (the terminal-NUL variant writes the NUL explicitly).
+	nameUnits := utf16.Encode([]rune(name))
+	shortUnits := utf16.Encode([]rune(shortName))
+	size := fileIDBothDirInfoFileNameOffset + len(nameUnits)*2
+	padded := (size + 7) &^ 7
+	buffer := make([]byte, padded)
+	binary.LittleEndian.PutUint32(buffer[0:4], next)
+	binary.LittleEndian.PutUint32(buffer[60:64], uint32(len(nameUnits)*2))
+	if len(shortUnits) > 0 {
+		buffer[68] = uint8(len(shortUnits) * 2)
+	}
+	for i, unit := range shortUnits {
+		binary.LittleEndian.PutUint16(buffer[70+i*2:72+i*2], unit)
+	}
+	binary.LittleEndian.PutUint64(buffer[96:104], fileID)
+	for i, unit := range nameUnits {
+		binary.LittleEndian.PutUint16(buffer[fileIDBothDirInfoFileNameOffset+i*2:fileIDBothDirInfoFileNameOffset+i*2+2], unit)
+	}
+	return buffer
+}

+// TestP84W12DirectoryEntryScannerValidation pins the focused hardening of the
+// directory-entry evidence scanner (scanFileIDBothDirInfoForShortName): a
+// malformed buffer - an invalid NextEntryOffset, an illegal UTF-16 length, or a
+// NUL inside a declared name range - refuses the admission fail-closed instead
+// of being parsed on a guess, while the ordinary shapes still evidence.
+//
+// Mutation checks: deleting the NextEntryOffset validation, the length checks,
+// or the embedded-NUL scans each turn the corresponding refusal assertion red.
+func TestP84W12DirectoryEntryScannerValidation(t *testing.T) {
+	const wantHigh, wantLow uint32 = 0x00000012, 0x00000034
+	fileID := uint64(wantHigh)<<32 | uint64(wantLow)
+
+	t.Run("valid-entry-with-short-name", func(t *testing.T) {
+		buffer := p84DirEntryBuffer("D", "R1PRO~1", fileID, 0)
+		shortName, evidenced, refuse := scanFileIDBothDirInfoForShortName(buffer, "D", wantHigh, wantLow)
+		if refuse || !evidenced || shortName != "R1PRO~1" {
+			t.Fatalf("the matching entry must evidence its short name, got %q evidenced=%v refuse=%v", shortName, evidenced, refuse)
+		}
+	})
+
+	t.Run("valid-entry-without-short-name", func(t *testing.T) {
+		buffer := p84DirEntryBuffer("D", "", fileID, 0)
+		shortName, evidenced, refuse := scanFileIDBothDirInfoForShortName(buffer, "D", wantHigh, wantLow)
+		if refuse || evidenced || shortName != "" {
+			t.Fatalf("an entry without a short name is no evidence, got %q evidenced=%v refuse=%v", shortName, evidenced, refuse)
+		}
+	})
+
+	t.Run("terminal-nul-in-short-name-range-is-the-terminator", func(t *testing.T) {
+		buffer := p84DirEntryBuffer("D", "R1PRO~1", fileID, 0)
+		// Count the terminator in the declared range, the N14 shape.
+		buffer[68] = 16
+		binary.LittleEndian.PutUint16(buffer[70+14:72+14], 0)

+		if refuse || !evidenced || shortName != "R1PRO~1" {

+		binary.LittleEndian.PutUint16(buffer[70+4:72+4], 0)
+		if shortName, evidenced, refuse := scanFileIDBothDirInfoForShortName(buffer, "D", wantHigh, wantLow); !refuse {
+			t.Fatalf("an embedded NUL unit in the declared short-name range must refuse, got %q evidenced=%v refuse=%v", shortName, evidenced, refuse)
+		}
+	})

+		// The NUL must sit INSIDE the declared two-unit name range.
+		binary.LittleEndian.PutUint16(buffer[fileIDBothDirInfoFileNameOffset+2:fileIDBothDirInfoFileNameOffset+4], 0)
+		if shortName, evidenced, refuse := scanFileIDBothDirInfoForShortName(buffer, "DD", wantHigh, wantLow); !refuse {
+			t.Fatalf("an embedded NUL unit in the declared file-name range must refuse, got %q evidenced=%v refuse=%v", shortName, evidenced, refuse)
+		}
+	})

+	t.Run("odd-file-name-length-refuses", func(t *testing.T) {

+		}
+	})
